### 1. 실행 명령
```python
python -c "
import sys

files = [
    r'04_허브앱\앱\index.html',
    r'04_허브앱\앱\data\teams_data.js',
    r'04_허브앱\_빌드\_build_prototype_v0_1.py'
]

targets = [
    '조장',
    '75%',
    '구글 드라이브',
    'notion',
    '노션',
    '소크라테스식 역질문',
    '소크라테스 AI 역질문',
    'AI 튜터가 소크라테스식',
    'Vercel',
    '배포됨'
]

total_found = 0
for fpath in files:
    with open(fpath, 'r', encoding='utf-8') as f:
        content = f.read()
    print(f'=== Checking {fpath} ===')
    for t in targets:
        count = content.count(t)
        if count > 0:
            print(f'  FOUND [{t}]: {count} occurrences')
            total_found += count
        else:
            print(f'  OK [{t}]: 0 occurrences')

print(f'\nTOTAL BANNED KEYWORDS FOUND: {total_found}')
if total_found > 0:
    sys.exit(1)
"
```
